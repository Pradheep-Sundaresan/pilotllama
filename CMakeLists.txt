cmake_minimum_required(VERSION 3.24)
project(inferno VERSION 0.1.0 LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(inferno_core src/core/request.cpp src/core/runtime.cpp src/core/batch_engine.cpp
                        src/backend/model_prompt.cpp src/backend/nemotron_prompt.cpp
                        src/metrics/metrics.cpp src/metrics/memory.cpp)
target_include_directories(inferno_core PUBLIC include)
target_compile_features(inferno_core PUBLIC cxx_std_20)
set_target_properties(inferno_core PROPERTIES CXX_EXTENSIONS NO)
target_link_libraries(inferno_core PUBLIC Threads::Threads)

add_executable(inferno src/cli/main.cpp)
set_target_properties(inferno PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED YES
    CXX_EXTENSIONS NO
)
target_compile_definitions(inferno PRIVATE INFERNO_VERSION="${PROJECT_VERSION}")
option(INFERNO_WITH_LLAMA "Link the existing pinned native llama.cpp build" OFF)
option(INFERNO_WITH_GRPC "Build pinned local protobuf/gRPC bindings and adapter" OFF)
if(INFERNO_WITH_GRPC)
    include(cmake/InfernoGrpc.cmake)
endif()
set(INFERNO_TEST_MODEL "" CACHE FILEPATH "Opt-in local GGUF for integration tests")
if(INFERNO_WITH_LLAMA)
    include(cmake/InfernoLlama.cmake)
    target_link_libraries(inferno PRIVATE inferno_llama_backend)
    target_compile_definitions(inferno PRIVATE INFERNO_WITH_LLAMA=1)
else()
    target_link_libraries(inferno PRIVATE inferno_core)
endif()
if(INFERNO_WITH_GRPC AND INFERNO_WITH_LLAMA)
    add_executable(inferno_rpc_server src/rpc/main.cpp)
    target_link_libraries(inferno_rpc_server PRIVATE inferno_rpc inferno_llama_backend)
    target_compile_options(inferno_rpc_server PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(inferno_native_batch_check tests/support/native_batch_check.cpp)
    target_link_libraries(inferno_native_batch_check PRIVATE inferno_llama_backend)
    target_compile_options(inferno_native_batch_check PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(inferno PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(inferno_core PRIVATE -Wall -Wextra -Wpedantic)
elseif(MSVC)
    target_compile_options(inferno PRIVATE /W4)
    target_compile_options(inferno_core PRIVATE /W4)
endif()

include(CTest)
if(BUILD_TESTING)
    set(gtest_source "${CMAKE_CURRENT_SOURCE_DIR}/third_party/googletest")
    if(NOT EXISTS "${gtest_source}/CMakeLists.txt")
        message(FATAL_ERROR "GoogleTest source missing. See docs/setup.md; download requires approval. Use -DBUILD_TESTING=OFF for a core-only build.")
    endif()
    find_package(Git REQUIRED)
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/googletest-revision.txt" gtest_pin)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${gtest_source}" rev-parse HEAD
                    OUTPUT_VARIABLE gtest_revision OUTPUT_STRIP_TRAILING_WHITESPACE
                    RESULT_VARIABLE gtest_git_result)
    if(NOT gtest_git_result EQUAL 0 OR NOT gtest_revision STREQUAL gtest_pin)
        message(FATAL_ERROR "GoogleTest source differs from the recorded revision")
    endif()
    set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(GTEST_HAS_ABSL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${gtest_source}" "${CMAKE_CURRENT_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
    add_executable(inferno_unit_tests tests/unit/request_test.cpp tests/unit/bounded_queue_test.cpp
                   tests/unit/runtime_test.cpp tests/unit/model_prompt_test.cpp tests/unit/nemotron_prompt_test.cpp tests/unit/metrics_test.cpp
                   tests/unit/memory_test.cpp tests/unit/batch_engine_test.cpp)
    target_link_libraries(inferno_unit_tests PRIVATE inferno_core GTest::gtest_main)
    set_target_properties(inferno_unit_tests PROPERTIES CXX_EXTENSIONS NO)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(inferno_unit_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    include(GoogleTest)
    gtest_discover_tests(inferno_unit_tests PROPERTIES TIMEOUT 20)
    if(INFERNO_WITH_GRPC)
        add_executable(inferno_proto_tests tests/unit/proto_test.cpp)
        target_link_libraries(inferno_proto_tests PRIVATE inferno_proto GTest::gtest_main)
        gtest_discover_tests(inferno_proto_tests TEST_PREFIX "proto." PROPERTIES TIMEOUT 20)
        add_executable(inferno_rpc_fixture tests/support/rpc_fixture.cpp)
        target_link_libraries(inferno_rpc_fixture PRIVATE inferno_rpc)
    endif()
    if(INFERNO_WITH_LLAMA AND INFERNO_TEST_MODEL)
        if(NOT EXISTS "${INFERNO_TEST_MODEL}")
            message(FATAL_ERROR "INFERNO_TEST_MODEL must be an existing local GGUF")
        endif()
        add_executable(inferno_integration_tests tests/integration/llama_backend_test.cpp)
        target_link_libraries(inferno_integration_tests PRIVATE inferno_llama_backend GTest::gtest_main)
        add_test(NAME native_model_integration COMMAND inferno_integration_tests)
        set_tests_properties(native_model_integration PROPERTIES
            TIMEOUT 120 LABELS "model;integration"
            ENVIRONMENT "INFERNO_TEST_MODEL=${INFERNO_TEST_MODEL}")
    endif()
endif()
